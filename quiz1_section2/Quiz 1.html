coord_x = [1,4, 0, -1, 20, 40]
coord_y = [2,2, 0, 10, 0, 1]

### Your solution below
hypotenuse_sqr = np.zeros(len(coord_x))
idx = 0
for x, y in zip(coord_x, coord_y):
    hypotenuse_sqr[idx] = x**2 + y**2
    idx = idx+1
plt.hist(hypotenuse_sqr, color="purple")
plt.xlabel("Squared Hypotenuses")
plt.ylabel("X and Y Coordinates")
plt.title("Frequency of Squared Hypotenuses")
plt.show()
